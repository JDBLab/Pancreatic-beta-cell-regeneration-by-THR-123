macro "Batch MFI & Counts Analysis" {
    baseDir = "C:/Users/mad1188/Box/THR in vivo 2023/Working Folder Layer 0/Processed";
    run("Set Measurements...", "area mean integrated redirect=None decimal=3");

    // your threshold values
    redMin   = 202; redMax   = 255;   // CK7
    greenMin = 134; greenMax = 255;   // BrdU
    andMin   = 155; andMax   = 255;   // CK7+BrdU colocalization

    subs = getFileList(baseDir);
    for (i=0; i<subs.length; i++) {
        name   = subs[i];
        folder = baseDir + "/" + name;
        if (!File.isDirectory(folder)) continue;

        comp = folder + "/Composite.tif";
        mask = folder + "/weka_mask.tif";
        if (!File.exists(comp) || !File.exists(mask)) continue;

        // —— split channels ——  
        open(comp);
        run("Split Channels");  // C1=CK7, C2=BrdU, C3=DAPI, C4=Insulin

        // —— open mask for ANDs & MFI ROI ——  
        open(mask);
        rename("mask");

        // —— CK7⁺ count ——  
        selectWindow("C1");
        setThreshold(redMin, redMax, "raw");
        setOption("BlackBackground", true);
        run("Convert to Mask");
        run("Analyze Particles...", "size=50-Infinity show=Nothing clear summarize");
        saveAs("Results", folder + "/CK7_counts.csv");
        close();  // CK7 mask

        // —— BrdU⁺ count ——  
        selectWindow("C2");
        setThreshold(greenMin, greenMax, "raw");
        setOption("BlackBackground", true);
        run("Convert to Mask");
        run("Analyze Particles...", "size=50-Infinity show=Nothing clear summarize");
        saveAs("Results", folder + "/BrdU_counts.csv");
        close();  // BrdU mask

        // —— CK7+BrdU colocalization count ——  
        imageCalculator("AND create", "C1", "C2");
        rename("and_mask");
        selectWindow("and_mask");
        setThreshold(andMin, andMax, "raw");
        setOption("BlackBackground", true);
        run("Convert to Mask");
        run("Analyze Particles...", "size=50-Infinity show=Nothing clear summarize");
        saveAs("Results", folder + "/CK7_Brdu_AND_counts.csv");
        close("and_mask");

        // ——  CK7 MFI in insulin regions ——  
        selectWindow("mask");  // if you want to use the Weka mask as ROI
        run("Create Selection");
        selectWindow("C1");    // CK7 channel
        run("Measure");
        saveAs("Results", folder + "/CK7_MFI_in_insulin.csv");

        // cleanup
        close("mask");
        close("C1"); close("C2"); close("C3"); close("C4");
        close(comp);
    }
    print("MFI & counts analysis done!");
}
